cmake_minimum_required(VERSION 3.20)
project(folia_soloist_capture LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Process-loopback capture requires Windows")
endif()
add_executable(folia-soloist-capture main.cpp)
target_compile_features(folia-soloist-capture PRIVATE cxx_std_17)
target_compile_definitions(folia-soloist-capture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(folia-soloist-capture PRIVATE ole32 mmdevapi)
if(MSVC)
  target_compile_options(folia-soloist-capture PRIVATE /W4 /EHsc)
endif()
set_target_properties(folia-soloist-capture PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/../../build/soloist-capture")
foreach(CONFIG Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER "${CONFIG}" CONFIG_UPPER)
  set_target_properties(folia-soloist-capture PROPERTIES
    "RUNTIME_OUTPUT_DIRECTORY_${CONFIG_UPPER}" "${CMAKE_SOURCE_DIR}/../../build/soloist-capture")
endforeach()
